# imgsizes.py
import json, os, sys, zlib, gzip
root = "/root/sfab/images/registry-data"
blobs = os.path.join(root, "docker/registry/v2/blobs/sha256")
def blob(d):
    h = d.split(":", 1)[1]
    return os.path.join(blobs, h[:2], h, "data")
def load(d):
    return json.load(open(blob(d)))
def unpacked(d, mt):
    p = blob(d)
    if "gzip" in mt or mt.endswith("tar+gzip") or mt.endswith(".tar.gzip"):
        n = 0
        with gzip.open(p, "rb") as f:
            while True:
                b = f.read(1 << 20)
                if not b: break
                n += len(b)
        return n
    if "zstd" in mt:
        return None
    return os.path.getsize(p)
seed = json.load(open(os.path.join(root, "SEED.json")))
tot_c = tot_u = 0
for im in seed["images"]:
    m = load(im["served"])
    if "manifests" in m:
        sel = [x for x in m["manifests"] if (x.get("platform") or {}).get("os") == "linux" and (x.get("platform") or {}).get("architecture") == "amd64"]
        if not sel:
            print("NOAMD64", im["id"]); continue
        m = load(sel[0]["digest"])
    c = m["config"]["size"] + sum(l["size"] for l in m["layers"])
    u = 0
    for l in m["layers"]:
        x = unpacked(l["digest"], l["mediaType"])
        u = None if (x is None or u is None) else u + x
    tot_c += c
    if u is not None: tot_u += u
    print(f'{im["id"]}\t{im["mirror"]}\tcompressed={c/2**20:.1f}MiB\tunpacked={"?" if u is None else f"{u/2**20:.1f}MiB"}')
print(f"TOTAL compressed={tot_c/2**30:.2f}GiB unpacked={tot_u/2**30:.2f}GiB")
